In the staggered-fermion formulation, we consider finite one-dimensional lattice of $N$ sites with fermionic variables $\chi_n$, normalized as $\{\chid_n, \chi_m\}=\dl_{nm}$, at each lattice site and $N$ is even. Here, the sites are indexed as $n=0,1,\ldots,N-1$ with discretized lattice spacing $a$. We consider a system with open boundary conditions.


Electric field operator,
\eq
L_n - L_{n-1} = \chid_n \chi_{n} - \frac{1 + (-1)^n}{2} + Q^+_n(t) - Q^-_n(t),
\eqx
with $L_{-1} = 0$ and $L_{N-1} = 0$ in a state constrained to $N/2$ fermions and zero total charge.
Two moving charges with trajectories, $x=\pm v t$ are smeared across neighboring sites as
\eq
Q^\pm_n(t) = Q\cdot \max\left( 1- |n_0 \pm vt/a -n|, 0 \right).
\eqx

Components of energy-momentum tensor are discretized as
\eqn
T^{00}_n &=& \f{i}{4a^2}  (K_n + K_{n-1}) + \f{g^2}{4} (L_n^2 +  L_{n-1}^2) +\f{m}{a} (-1)^n \chid_n \chi_n, \nonumber \\
T^{11}_n &=& \f{i}{4a^2}  (K_n + K_{n-1}) - \f{g^2}{4} (L_n^2 +  L_{n-1}^2),  \nonumber\\
T^{01}_n &=& \f{i}{4a^2} \left( \chid_{n+1} \chi_{n-1} -  \chid_{n-1} \chi_{n+1}\right),
\eqnx
where the kinetic term component $K_n = \chid_{n+1} \chi_{n} - \chid_n \chi_{n+1}$ and $K_{-1} = K_{N-1} = 0$.

\noindent The Hamiltonian reads
\eqn
\label{H00}
\mathbb H = a \sum_{n=0}^{N-1} T^{00}_n = \f{i}{2a} \sum_{n=0}^{N-2} \left( \chid_{n+1} \chi_{n} - \chid_n \chi_{n+1} \right) +  m \sum_{n=0}^{N-1} (-1)^n \chid_n \chi_n +  \f{g^2 a}{2} \sum_{n=0}^{N-1} L_n^2
\eqnx

\noindent  Centering in the middle of the lattice $n_0 = (N-1)/2$, the boost operator reads
\eqn
\label{H00}
\mathbb K &=& -a^2 \sum_{n=0}^{N-1} (n-n_0) T^{00}_n  \nonumber \\
                  &=&  \f{-i}{2} \sum_{n=0}^{N-1} \left(n +\frac12 - n_0 \right) \left( \chid_{n+1} \chi_{n} - \chid_n \chi_{n+1} \right)  -  a m \sum_{n=0}^{N-1} (-1)^n \left(n - n_0 \right) \chid_n \chi_n -  \f{g^2 a^2}{2} \sum_{n=0}^{N-1} \left(n + \frac12 - n_0 \right) L_n^2
\eqnx


We define current operators and chiral condensate density for pairs or sites
\eqn
j^{0}_i  &=& \frac{1}{2a} \left(\chid_{2i} \chi_{2i} + \chid_{2i+1} \chi_{2i+1} \right),  \nonumber\\
j^{1}_i &=&  \frac{1}{2a} \left(\chid_{2i} \chi_{2i+1} + \chid_{2i+1} \chi_{2i} \right),  \nonumber\\
\nu_i  &=& \frac{1}{2a} \left(\chid_{2i} \chi_{2i} - \chid_{2i+1} \chi_{2i+1} \right),  
\eqnx


\newpage

We boost the first excited state $|P=0\rangle$ as
$$
|P \rangle = e^{-i \chi \mathbb K} |P=0\rangle
$$
where $\chi$ follows as
$$
P = E \frac{e^{2\chi} - 1}{e^{2\chi} + 1},
$$
and $E$ is the energy above the ground state.
